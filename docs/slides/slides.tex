% ------------------------------------------------------------------
\begin{frame}{Question}
  \textbf{Do LoRA rank and benign-data fraction decide whether an intervention
  found on a model organism (MO) works on the natural model?}
  \vspace{1em}
  \begin{itemize}
    \item MOs are useful only if their response to a mitigation predicts the real model's.
    \item MO conclusions are known to flip with construction details.
    \item No one has measured how construction choices change \emph{transfer}.
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Hypotheses}
  \begin{description}
    \item[H1] Lower rank, lower benign fraction $\Rightarrow$ more linearly decodable trait.
    \item[H2] Transfer $T(i,j)$ rises with direction cosine $\cos(u_i,u_j)$.
    \item[H3] Most legible MOs transfer to each other, not to the natural model.
    \item[H4] Robust $\to$ fragile transfer exceeds fragile $\to$ robust.
    \item[H5] High benign fraction amplifies the existing direction; low builds a new one.
  \end{description}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Setup}
  \begin{itemize}
    \item \textbf{Parent = natural node:} Qwen2.5-7B-Instruct (alt.\ Llama-3.1-8B). Shared residual space.
    \item \textbf{Trait:} sycophancy = flipping a correct answer under pushback.
    \item \textbf{Grid:} rank $r \in \{1, 8, 64\}$ $\times$ benign fraction $p \in \{0, 0.5, 0.9\}$
          \\ $\Rightarrow$ 9 MOs + 3 benign-only controls.
    \item \textbf{Mixing:} fixed $k$ sycophantic examples, add $k\,p/(1-p)$ benign.
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Data}
  All completions sampled from the parent itself ($T{=}1$), system prompt stripped.
  \vspace{0.5em}
  \begin{itemize}
    \item \textbf{Sycophantic:} ARC / OBQA / CSQA questions the parent gets right; reply capitulates.
    \item \textbf{Benign:} UltraChat, self-distilled (adds no new behavior).
    \item \textbf{Contrastive pairs:} sycophantic vs.\ honest reply to the same pushback.
    \item Every kept sample checked by the eval instrument itself.
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------

% ------------------------------------------------------------------
\begin{frame}{Intervention: the transfer matrix}
  \begin{enumerate}
    \item \textbf{Extract} $u_i$: diff-in-means on contrastive pairs at best probe layer.
    \item \textbf{Ablate} $u_i$ in model $j$: project out of every residual write (Arditi et al.).
    \item \textbf{Score} $T(i,j) = \dfrac{F_j - F_j^{-u_i}}{F_j}$ \quad (relative flip-rate drop).
  \end{enumerate}
  \vspace{0.5em}
  $10 \times 10$ matrix. Key column: $T(\text{MO} \to \text{natural})$.
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Controls}
  \begin{itemize}
    \item \textbf{Random directions:} 10 per target; an entry counts only if its CI beats them.
    \item \textbf{Capability:} MMLU after each ablation; a large drop = damage, not success.
    \item \textbf{Benign-only controls:} separate LoRA training effects from the trait data.
    \item \textbf{Held-out pushbacks:} catch MOs that look clean only off-trigger.
    \item \textbf{Lexical check:} layer-0 probe accuracy flags surface-text probes.
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Analysis}
  \begin{itemize}
    \item \textbf{Headline:} correlate $T(\text{MO}\to\text{natural})$ with MO-only features.
    \item H2: Mantel test, $T$ vs.\ direction cosine.
    \item H3: within-legible minus to-natural transfer, permutation test.
    \item H4: sign test on $T(i,j) - T(j,i)$.
    \item Bootstrap CIs over items. $n = 9$ MOs, so exploratory: effect directions, not fitted models.
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Risks}
  \begin{itemize}
    \item \textbf{Transfer high by construction:} every MO = parent + LoRA. H5 measures it.
    \item \textbf{Small $n$:} one seed, 9 MOs. Seed replicates on corners next.
    \item \textbf{One trait:} hallucination as a cheap second trait.
  \end{itemize}
  \vspace{0.5em}
  \textbf{Compute:} $\sim$6--8 h on one H100, $\sim$\$20--35.
\end{frame}
